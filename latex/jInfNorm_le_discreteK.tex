\section{Setting and statement}

Let \(\alpha\) be an additive commutative monoid, \(M\) an additive commutative monoid with a
preorder \(\le\), and \(x \mapsto \abs{x}\) a map \(\alpha \to M\) (\texttt{Modulus}) satisfying
\(\abs{0} = 0\), \(0 \le \abs{x}\),
\begin{equation}\label{eq:abs-add}
  \abs{x+y} \le \abs{x} + \abs{y} \qquad (x, y \in \alpha),
\end{equation}
and the following \emph{refined Riesz decomposition}:
\begin{itemize}
  \item[(R)] if \(\abs{x} \le \abs{y} + \abs{z}\), then there are \(x_0, x_1 \in \alpha\) with
  \[
    x = x_0 + x_1,\qquad \abs{x_0} \le \abs{y},\quad \abs{x_1} \le \abs{z},\qquad
    \abs{x_0} \le \abs{x},\quad \abs{x_1} \le \abs{x}.
  \]
\end{itemize}
These four conditions are the fields \texttt{modulus\_zero}, \texttt{modulus\_nonneg},
\texttt{modulus\_add\_le} and \texttt{exists\_decomp} of \texttt{Modulus.IsDecomposable}. The
first two give \(\abs{0} \le \abs{x}\) (\texttt{modulus\_zero\_le}). The instances are the
following.
\begin{itemize}
  \item On \(\mathbb R\) and \(\mathbb C\), \(M = [0,\infty]\), \(\abs{x}\) is the norm, and
    (R) holds with
    \(x_0 = cx\), \(x_1 = (1-c)x\), \(c = \min(1, \abs{y}/\abs{x})\)
    (\texttt{Modulus.isDecomposable\_ofENorm}, for every real normed space).
  \item On \([0,\infty]\), \(M = [0,\infty]\), \(\abs{x} = x\), \(x_0 = \min(x, y)\) and
    \(x_1 = x - x_0\).
  \item On products, \(M\) is the product of the value monoids, \(\abs{\cdot}\) is taken
    pointwise, and (R) is checked pointwise.
\end{itemize}

Let \(A_0, A_1\) be quasinorms on \(\alpha\) with values in \([0,\infty]\) that are
\emph{solid}: \(\abs{x} \le \abs{y}\) implies \(\norm{x}{i} \le \norm{y}{i}\), where
\(\norm{\cdot}{i}\) is the quasinorm of \(A_i\). No other property of \(A_0, A_1\) is used; in
particular their quasi-triangle constants never appear.

Fix \(0 < \theta < 1\) and \(0 < q \le \infty\). For \(\nu \in \Z\), \(x, u \in \alpha\) put
\[
  K_\nu(x) = \inf_{x = x_0 + x_1} \bigl(\norm{x_0}{0} + 2^\nu \norm{x_1}{1}\bigr),
  \qquad
  J_\nu(u) = \max\bigl(\norm{u}{0},\ 2^\nu \norm{u}{1}\bigr),
\]
and \(\Delta = \{x \in \alpha : \norm{x}{0} < \infty,\ \norm{x}{1} < \infty\}\).
For a family \(f = (f_\nu)_{\nu \in S}\) in \([0,\infty]\) indexed by a countable set \(S\),
\(\norm{f}{\ell^q}\) is \((\sum_{\nu\in S} f_\nu^q)^{1/q}\) if \(q < \infty\) and
\(\sup_{\nu \in S} f_\nu\) if \(q = \infty\). The two norms to be compared are
\[
  \norm{x}{K} = \bigl\lVert (2^{-\nu\theta} K_\nu(x))_{\nu \in \Z} \bigr\rVert_{\ell^q}
  \qquad (\texttt{discreteKMethod}),
\]
\[
  \cJ(x) = \inf \Bigl\{ \bigl\lVert (2^{-\nu\theta} J_\nu(u_\nu))_{\nu \in \Z} \bigr\rVert_{\ell^q}
  : u \colon \Z \to \alpha \text{ finitely supported},\ \textstyle\sum_{\nu \in \Z} u_\nu = x \Bigr\}
  \qquad (\texttt{jInfNorm}).
\]
Since \(J_\nu(0) = 0\), a family \(u \colon \{-N, \dots, N-1\} \to \alpha\) with \(\sum_{\nu=-N}^{N-1} u_\nu = x\), extended by zero to \(\Z\), is
admissible, with cost \(\lVert (2^{-\nu\theta} J_\nu(u_\nu))_{\nu=-N}^{N-1} \rVert_{\ell^q}\).
Conversely every admissible \(u\) is supported in \(\{-N, \dots, N-1\}\) for some \(N\). Hence \(\cJ\) is equivalently the
infimum of these costs over \(N \in \mathbb N\) and \(u \colon \{-N, \dots, N-1\} \to \alpha\).

\begin{theorem}\label{thm:main}
  For every \(x \in \Delta\),
  \[
    \cJ(x) \le 2^{\min(\theta,\, 1-\theta)}\, \norm{x}{K} .
  \]
\end{theorem}

The constant is at most \(\sqrt 2\) and does not depend on \(q\) or on the couple.

\paragraph{Idea of the proof.}
For each level \(\mu\) choose a near-optimal decomposition \(x = y_\mu + z_\mu\) for
\(K_\mu(x)\). Using (R), split \(x\) into \(2N\) pieces, the piece \(u_\nu\) being dominated
by \(y_{\nu+1}\) and by \(z_\nu\). Then \(J_\nu(u_\nu) \lesssim K_{\nu+1}(x)\), except for one
of the two norms of each of the two end pieces, which is controlled only by \(x\) itself.
After weighting, these end terms decay geometrically in \(N\), because \(0<\theta<1\) and
\(x \in \Delta\). The \(\ell^q\)-norm is estimated through its \(r\)-th power,
\(r = \min(q,1)\), which is subadditive with constant \(1\). The error terms vanish in the
limit, so the main term keeps the constant \(2^\theta\) for every \(q\).
Swapping \(A_0\) and \(A_1\) replaces \(\theta\) by \(1-\theta\) in both norms, so the bound
with \(2^{1-\theta}\) follows from the bound with \(2^\theta\) for the swapped couple.

\section{Elementary facts}

\begin{lemma}\label{lem:K-dyadic}
  For every \(x \in \alpha\) and \(\nu \in \Z\),
  \(K_\nu(x) \le K_{\nu+1}(x)\).
\end{lemma}

\begin{proof}
  For any decomposition \(x = x_0 + x_1\),
  \(\norm{x_0}{0} + 2^\nu\norm{x_1}{1} \le \norm{x_0}{0} + 2^{\nu+1}\norm{x_1}{1}\).
  Taking the infimum over all decompositions gives the claim.
\end{proof}

The same argument shows that \(t \mapsto \inf_{x = x_0 + x_1}\bigl(\norm{x_0}{0} +
t\norm{x_1}{1}\bigr)\) is monotone on \([0,\infty]\) for every \(x \in \alpha\); in Lean this is
\texttt{monotone\_kNorm}. The lemma is its case \(2^\nu \le 2^{\nu+1}\)
(\texttt{ENNReal.zpow\_le\_of\_le}).

\begin{lemma}\label{lem:near-opt}
  For every \(x \in \alpha\), \(\mu \in \Z\) and \(\delta > 0\) there is a decomposition
  \(x = y + z\) with \(\norm{y}{0} + 2^\mu \norm{z}{1} \le K_\mu(x) + \delta\).
\end{lemma}

\begin{proof}
  If \(K_\mu(x) = \infty\), any decomposition works, e.g.\ \(x = 0 + x\). Otherwise
  \(K_\mu(x) < K_\mu(x) + \delta\), and by the definition of the infimum some decomposition
  has cost below \(K_\mu(x) + \delta\) (\texttt{exists\_decomp\_lt\_of\_lt\_kNorm}).
\end{proof}

An additive error is used because the infimum need not be attained, and a multiplicative error
\(1 + \varepsilon\) is useless when \(K_\mu(x) = 0\).

\begin{lemma}\label{lem:lq-sub}
  Put \(r = \min(q,1)\). For \(f, g \colon \Z \to [0,\infty]\),
  \[
    \norm{f+g}{\ell^q}^r \le \norm{f}{\ell^q}^r + \norm{g}{\ell^q}^r .
  \]
\end{lemma}

\begin{proof}
  If \(q \ge 1\), including \(q = \infty\), then \(r = 1\) and this is Minkowski's inequality.
  Let \(0 < q < 1\), so that \(r = q\) and \(\norm{f}{\ell^q}^q = \sum_{\nu \in \Z} f_\nu^q\).
  For \(s, t \in [0,\infty]\) we have \((s+t)^q \le s^q + t^q\). This is clear if \(s+t\) is
  \(0\) or \(\infty\); otherwise it follows from \(\lambda^q \ge \lambda\) for
  \(\lambda \in [0,1]\), applied to \(\lambda = s/(s+t)\) and \(\lambda = t/(s+t)\). Summing
  this inequality with \(s = f_\nu\), \(t = g_\nu\) over \(\nu \in \Z\) gives the claim.
\end{proof}

In Mathlib, Lemma~\ref{lem:lq-sub} is \texttt{inhmgELpNorm\_add\_le}. For \(q \ne 0\),
\(\norm{f}{\ell^q}^r\) equals \texttt{inhmgELpNorm f q Measure.count}, by
\begin{itemize}
  \item \texttt{inhmgELpNorm\_eq\_eLpNorm} if \(q \ge 1\),
  \item \texttt{inhmgELpNorm\_eq\_eLpNorm\_pow} if \(q \le 1\).
\end{itemize}

\begin{lemma}\label{lem:lq-facts}
  Let \(f, g \colon \Z \to [0,\infty]\).
  \begin{enumerate}
    \item[(a)] If \(c \in [0,\infty)\) and \(f_\nu \le c\, g_\nu\) for all \(\nu \in \Z\), then
      \(\norm{f}{\ell^q} \le c \norm{g}{\ell^q}\).
    \item[(b)] For every bijection \(\sigma\) of \(\Z\),
      \(\norm{(f_{\sigma(\nu)})_{\nu \in \Z}}{\ell^q} = \norm{f}{\ell^q}\).
    \item[(c)] If \(S \subseteq \Z\) is finite with \(n\) elements and \(c \in [0,\infty]\), then
      \(\norm{c\, \one_S}{\ell^q} \le c\, n^{1/q}\), where \(1/\infty = 0\).
  \end{enumerate}
\end{lemma}

\begin{proof}
  (a) The \(\ell^q\)-norm is monotone and positively homogeneous. (b) Sums and suprema are
  invariant under reindexing by a bijection. (c) If \(q < \infty\), then \(\norm{c\,\one_S}{\ell^q}^q = n c^q\). If
  \(q = \infty\), then \(\norm{c\,\one_S}{\ell^\infty}\) is \(c\) or \(0\), hence at most
  \(c = c\, n^0\).
\end{proof}

In Lean:
\begin{itemize}
  \item[(a)] \texttt{eLpNorm\_le\_mul\_eLpNorm\_of\_ae\_le\_mul'} with \(c : \mathbb R_{\ge 0}\);
  \item[(b)] \texttt{eLpNorm\_comp\_measurePreserving} with\\
    \texttt{Function.Bijective.measurePreserving\_count};
  \item[(c)] \texttt{eLpNorm\_indicator\_const\_le} with \texttt{Measure.count\_apply\_finset}.
\end{itemize}
The lemma \texttt{Function.Bijective.measurePreserving\_count} (a measurable bijection
preserves \texttt{Measure.count}) is not in Mathlib; it is added in this project.
Every function on \(\Z\) is measurable and a.e.\ strongly measurable, by
\texttt{Measurable.of\_discrete} and \texttt{AEStronglyMeasurable.of\_discrete}.

\section{Splitting \(x\) into \(2N\) pieces}

This is the only place where (R) is used. The first piece is split off \(r\), and the rest
of \(r\) is split by induction, so no subtraction is needed.

\begin{proposition}\label{prop:split-general}
  Let \(x \in \alpha\), \(a \in \Z\), \(m \in \mathbb N\), put \(I = \{a, \dots, a+m\}\), and
  for \(\mu = a+1, \dots, a+m\) let \(x = y_\mu + z_\mu\) be arbitrary decompositions. Let
  \(r \in \alpha\) with \(\abs{r} \le \abs{x}\). Then there is \(u \colon \Z \to \alpha\) with
  \(u_\nu = 0\) for \(\nu \notin I\), \(\sum_{\nu \in \Z} u_\nu = r\) and, for every
  \(\nu \in I\),
  \begin{enumerate}
    \item[(a)] \(\abs{u_\nu} \le \abs{y_{\nu+1}}\) if \(\nu < a+m\);
    \item[(b)] \(\abs{u_\nu} \le \abs{z_\nu}\) if \(\nu > a\);
    \item[(c)] \(\abs{u_\nu} \le \abs{r}\).
  \end{enumerate}
\end{proposition}

\begin{proof}
  By induction on \(m\), for all \(a\) and \(r\) simultaneously.

  \emph{The case \(m = 0\).} Let \(u_a = r\) and \(u_\nu = 0\) for \(\nu \ne a\). Then
  (a) and (b) are vacuous, and (c) holds by reflexivity.

  \emph{The step from \(m\) to \(m+1\).} Since \(x = y_{a+1} + z_{a+1}\), inequality
  \eqref{eq:abs-add} and transitivity give
  \[
    \abs{r} \le \abs{x} \le \abs{y_{a+1}} + \abs{z_{a+1}} .
  \]
  Applying (R) to \(r\) gives \(r = v + r'\) with
  \begin{equation}\label{eq:step}
    \abs{v} \le \abs{y_{a+1}}, \qquad \abs{r'} \le \abs{z_{a+1}}, \qquad
    \abs{v} \le \abs{r}, \qquad \abs{r'} \le \abs{r}.
  \end{equation}
  By the last inequality, \(\abs{r'} \le \abs{x}\), so the induction hypothesis applies to
  the start \(a+1\), the length \(m\) and the remainder \(r'\). It gives
  \(u' \colon \Z \to \alpha\) vanishing outside \(I' = \{a+1, \dots, a+1+m\}\) with
  \(\sum_{\nu \in \Z} u'_\nu = r'\) and (a)--(c) relative to \(I'\) and \(r'\). Define
  \(u_a = v\) and \(u_\nu = u'_\nu\) for \(\nu \ne a\). Then \(u\) vanishes outside
  \(I = \{a\} \cup I'\), and since \(u'_a = 0\), \(\sum_{\nu \in \Z} u_\nu = v + \sum_{\nu \in \Z} u'_\nu = v + r' = r\).

  At \(\nu = a\), (a) and (c) are the first and third inequality of \eqref{eq:step}, and (b)
  is vacuous. Let \(\nu \in I'\), so \(u_\nu = u'_\nu\). Property (a) is that of \(u'\), since
  the right end of \(I'\) is that of \(I\). Property (c) follows from
  \(\abs{u'_\nu} \le \abs{r'} \le \abs{r}\). For (b), if \(\nu > a+1\) it is that of \(u'\).
  At \(\nu = a+1\) the induction hypothesis gives no condition (b), because \(a+1\) is the left
  end of \(I'\); instead, (c) for \(u'\) and the second inequality of \eqref{eq:step} give
  \(\abs{u'_{a+1}} \le \abs{r'} \le \abs{z_{a+1}}\).
\end{proof}

Both new conditions of (R) are needed. The condition \(\abs{x_1} \le \abs{x}\) gives
\(\abs{r'} \le \abs{x}\), without which the induction hypothesis could not be applied; the
condition \(\abs{x_0} \le \abs{x}\), through (c), gives (b) at the left end of \(I'\). For
the same reason (c) is stated relative to \(r\) and not to \(x\).

In Lean, Propositions~\ref{prop:split-general} and~\ref{prop:split} are
\texttt{exists\_decomp\_Icc} and \texttt{exists\_decomp\_Ico\_symm}, with (a)--(c) stated for
every \(\nu \in \Z\). For \(\nu \notin I\) they hold because \(u_\nu = 0\) and
\(\abs{0} \le \abs{w}\) for every \(w \in \alpha\).

\begin{proposition}\label{prop:split}
  Let \(N \ge 1\), \(x \in \alpha\), and for \(\mu = -N+1, \dots, N-1\) let
  \(x = y_\mu + z_\mu\) be arbitrary decompositions. Then there is
  \(u \colon \Z \to \alpha\) vanishing outside \(\{-N, \dots, N-1\}\) with \(\sum_{\nu \in \Z} u_\nu = x\) and,
  for every \(\nu \in \{-N, \dots, N-1\}\),
  \begin{enumerate}
    \item[(a)] \(\abs{u_\nu} \le \abs{y_{\nu+1}}\) if \(\nu \le N-2\);
    \item[(b)] \(\abs{u_\nu} \le \abs{z_\nu}\) if \(\nu \ge -N+1\);
    \item[(c)] \(\abs{u_\nu} \le \abs{x}\).
  \end{enumerate}
\end{proposition}

\begin{proof}
  Apply Proposition~\ref{prop:split-general} with \(a = -N\), \(m = 2N-1\) and \(r = x\);
  then \(I = \{-N, \dots, N-1\}\).
\end{proof}

\section{The pointwise estimate}

\begin{lemma}\label{lem:pointwise}
  Let \(x \in \alpha\), \(\delta > 0\) and \(N \ge 1\). For \(\mu = -N+1, \dots, N-1\) choose
  decompositions \(x = y_\mu + z_\mu\) as in Lemma~\ref{lem:near-opt}, i.e.
  \begin{equation}\label{eq:near-opt}
    \norm{y_\mu}{0} + 2^\mu \norm{z_\mu}{1} \le K_\mu(x) + \delta,
  \end{equation}
  and let \(u \colon \Z \to \alpha\) be given by Proposition~\ref{prop:split}. Put
  \(E_{-N} = 2^{-N}\norm{x}{1}\), \(E_{N-1} = \norm{x}{0}\) and \(E_\nu = 0\) for the other
  \(\nu \in \{-N, \dots, N-1\}\). Then for every \(\nu \in \{-N, \dots, N-1\}\),
  \[
    J_\nu(u_\nu) \le \max\bigl(K_{\nu+1}(x) + \delta,\ E_\nu\bigr).
  \]
\end{lemma}

\begin{proof}
  Since \(N \ge 1\), the indices \(-N\) and \(N-1\) are distinct, so each \(\nu\) is
  exceptional in at most one of the two entries of \(J_\nu(u_\nu)\). We bound both entries
  by the right-hand side; solidity turns each modulus inequality of
  Proposition~\ref{prop:split} into an inequality of quasinorms.

  \emph{The entry \(\norm{u_\nu}{0}\).} If \(\nu \le N-2\), then by (a) and
  \eqref{eq:near-opt} at \(\mu = \nu+1\),
  \(\norm{u_\nu}{0} \le \norm{y_{\nu+1}}{0} \le K_{\nu+1}(x) + \delta\).
  If \(\nu = N-1\), then by (c), \(\norm{u_\nu}{0} \le \norm{x}{0} = E_{N-1}\).

  \emph{The entry \(2^\nu \norm{u_\nu}{1}\).} If \(\nu \ge -N+1\), then by (b),
  \eqref{eq:near-opt} at \(\mu = \nu\), and Lemma~\ref{lem:K-dyadic},
  \(2^\nu\norm{u_\nu}{1} \le 2^\nu\norm{z_\nu}{1} \le K_\nu(x) + \delta \le K_{\nu+1}(x) +
  \delta\). If \(\nu = -N\), then by (c), \(2^{-N}\norm{u_\nu}{1} \le 2^{-N}\norm{x}{1} =
  E_{-N}\).
\end{proof}

\section{The bound with \(2^\theta\)}

\begin{proposition}\label{prop:two-theta}
  For every \(x \in \Delta\), \(\cJ(x) \le 2^\theta \norm{x}{K}\).
\end{proposition}

Fix \(x \in \Delta\), and put \(r = \min(q,1)\) and
\(a_\mu = 2^{-\mu\theta} K_\mu(x)\) for \(\mu \in \Z\), so that
\(\norm{a}{\ell^q} = \norm{x}{K}\). Fix \(N \ge 1\) and \(\delta > 0\), let \(u\) be as in
Lemma~\ref{lem:pointwise}, and put
\[
  b_\nu = 2^{-\nu\theta} J_\nu(u_\nu) \quad (\nu \in \Z), \qquad
  e_\nu = 2^{-\nu\theta} E_\nu \quad (\nu \in \{-N, \dots, N-1\}),
\]
and \(e_\nu = 0\) for the other \(\nu \in \Z\). The end terms are
\[
  e_{-N} = 2^{N\theta} 2^{-N}\norm{x}{1} = 2^{-(1-\theta)N}\norm{x}{1},
  \qquad
  e_{N-1} = 2^{-(N-1)\theta}\norm{x}{0}.
\]
Since \(u\) vanishes outside \(\{-N, \dots, N-1\}\) and \(\sum_{\nu \in \Z} u_\nu = x\), the
family \(u\) is admissible in the definition of \(\cJ\), so \(\cJ(x) \le \norm{b}{\ell^q}\).

\begin{lemma}\label{lem:termwise}
  For every \(\nu \in \Z\),
  \[
    b_\nu \le 2^\theta a_{\nu+1} + \delta\, 2^{N\theta}\, \one_{\{-N, \dots, N-1\}}(\nu) + e_\nu .
  \]
\end{lemma}

\begin{proof}
  Let \(\nu \in \{-N, \dots, N-1\}\). Bounding the maximum in Lemma~\ref{lem:pointwise} by the
  sum gives \(J_\nu(u_\nu) \le K_{\nu+1}(x) + \delta + E_\nu\), and we multiply this by
  \(2^{-\nu\theta}\). Since \(2^{-\nu\theta} = 2^\theta\, 2^{-(\nu+1)\theta}\), the first term
  becomes \(2^\theta a_{\nu+1}\). Since \(\nu \ge -N\) and \(\theta > 0\), we have
  \(2^{-\nu\theta} \le 2^{N\theta}\), which bounds the second term. The third term becomes
  \(e_\nu\).

  Let \(\nu \notin \{-N, \dots, N-1\}\). Then \(u_\nu = 0\), so
  \(b_\nu = 2^{-\nu\theta} J_\nu(0) = 0\), because quasinorms vanish at \(0\).
\end{proof}

The main term carries no indicator, so that Lemma~\ref{lem:lq-facts}(b) applies to it.

Since \(N \ge 1\), the indices \(-N\) and \(N-1\) are distinct, so
\(e = e_{-N}\one_{\{-N\}} + e_{N-1}\one_{\{N-1\}}\). By Lemma~\ref{lem:termwise} and
Lemma~\ref{lem:lq-facts}(a) with \(c = 1\), \(\norm{b}{\ell^q}\) is at most the
\(\ell^q\)-norm of the sum of the four families
\[
  (2^\theta a_{\nu+1})_{\nu \in \Z}, \qquad
  \delta\, 2^{N\theta}\, \one_{\{-N, \dots, N-1\}}, \qquad
  e_{-N}\one_{\{-N\}}, \qquad
  e_{N-1}\one_{\{N-1\}} .
\]
By Lemma~\ref{lem:lq-sub}, applied three times, \(\norm{b}{\ell^q}^r\) is at most the sum of
the \(r\)-th powers of their norms. By Lemma~\ref{lem:lq-facts}(a) and (b) with \(\sigma(\nu) = \nu + 1\), the
norm of the first family is at most \(2^\theta \norm{a}{\ell^q} = 2^\theta \norm{x}{K}\). By
Lemma~\ref{lem:lq-facts}(c), the norms of the other three are at most
\(\delta\, 2^{N\theta} (2N)^{1/q}\), \(e_{-N}\) and \(e_{N-1}\). Hence
\begin{equation}\label{eq:fixed-N}
  \cJ(x)^r \le 2^{\theta r}\norm{x}{K}^r + \bigl(\delta\, 2^{N\theta} (2N)^{1/q}\bigr)^r
    + e_{-N}^r + e_{N-1}^r .
\end{equation}

The left-hand side of \eqref{eq:fixed-N} depends on neither \(\delta\) nor \(N\). Fix \(N\)
and let \(\delta \to 0\). Since \(2^{N\theta} (2N)^{1/q}\) is finite and \(r > 0\), the second
term tends to \(0\), and since addition on \([0,\infty]\) is continuous,
\[
  \cJ(x)^r \le 2^{\theta r}\norm{x}{K}^r + e_{-N}^r + e_{N-1}^r .
\]
Now let \(N \to \infty\). Since \(\theta < 1\) and \(\norm{x}{1} < \infty\), \(e_{-N} \to 0\);
since \(\theta > 0\) and \(\norm{x}{0} < \infty\), \(e_{N-1} \to 0\). As \(r > 0\), their
\(r\)-th powers tend to \(0\) as well. This gives
\(\cJ(x)^r \le 2^{\theta r}\norm{x}{K}^r\), and taking \(r\)-th roots proves
Proposition~\ref{prop:two-theta}.
\qed

\medskip
The two limits are taken one after the other: \(\delta \to 0\) for fixed \(N\), then
\(N \to \infty\). This is legitimate because \(\cJ(x)\) depends on neither, and the end terms
do not depend on \(\delta\). No assumption \(\norm{x}{K} < \infty\) is needed, and no case
distinction between \(q < \infty\) and \(q = \infty\). In Lean the first limit can be taken
with \texttt{ge\_of\_tendsto'} along \(\delta = n^{-1}\), \(n \to \infty\)
(\texttt{ENNReal.tendsto\_inv\_nat\_nhds\_zero}); in \([0,\infty]\), \(n^{-1} \ne 0\) also for
\(n = 0\). The second limit needs \texttt{ge\_of\_tendsto} with
\texttt{Filter.eventually\_ge\_atTop 1}, since \eqref{eq:fixed-N} holds only for \(N \ge 1\).

\section{Symmetry, and the bound with \(2^{1-\theta}\)}

In this section the couple and \(\theta\) are written as arguments. For \(t \in [0,\infty]\)
and \(x, u \in \alpha\) put
\[
  K(t,x;A_0,A_1) = \inf_{x = x_0 + x_1}\bigl(\norm{x_0}{0} + t\norm{x_1}{1}\bigr),\qquad
  J(t,u;A_0,A_1) = \max\bigl(\norm{u}{0},\ t\norm{u}{1}\bigr)
\]
(\texttt{kNorm} and \texttt{jNorm}), so that \(K_\nu(x) = K(2^\nu,x;A_0,A_1)\) and
\(J_\nu(u) = J(2^\nu,u;A_0,A_1)\). We write \(K_\nu(x;A_0,A_1)\), \(J_\nu(u;A_0,A_1)\),
\(\norm{x}{K}(A_0,A_1;\theta)\) and \(\cJ(x;A_0,A_1;\theta)\) for the quantities of
Section~1. The swapped couple \((A_1,A_0)\) is \texttt{A.swap} (\texttt{Couple.swap}).
For \(g \colon \Z \to [0,\infty]\) put
\(\Phi_\theta(g) = \lVert (2^{-\nu\theta} g_\nu)_{\nu\in\Z} \rVert_{\ell^q}\)
(\texttt{discretePhiFunctional}). Then
\(\norm{x}{K}(A_0,A_1;\theta) = \Phi_\theta(\nu \mapsto K_\nu(x;A_0,A_1))\), and the cost of an
admissible family \(u\) in \(\cJ(x;A_0,A_1;\theta)\) is
\(\Phi_\theta(\nu \mapsto J_\nu(u_\nu;A_0,A_1))\).

\begin{lemma}\label{lem:swap-KJ}
  Let \(0 < t < \infty\) and \(x, u \in \alpha\). Then
  \begin{enumerate}
    \item[(a)] \(K(t,x;A_1,A_0) = t\,K(t^{-1},x;A_0,A_1)\);
    \item[(b)] \(J(t,u;A_1,A_0) = t\,J(t^{-1},u;A_0,A_1)\).
  \end{enumerate}
  In particular, for \(\nu \in \Z\),
  \[
    K_\nu(x;A_1,A_0) = 2^\nu K_{-\nu}(x;A_0,A_1),\qquad
    J_\nu(u;A_1,A_0) = 2^\nu J_{-\nu}(u;A_0,A_1).
  \]
\end{lemma}

\begin{proof}
  Since \(t \ne 0,\infty\), multiplication by \(t\) commutes with infima and with \(\max\) on
  \([0,\infty]\), and \(t\,t^{-1} = 1\). Hence \(t\,K(t^{-1},x;A_0,A_1)\) is the infimum of
  \(t\norm{x_0}{0} + \norm{x_1}{1}\) over all decompositions \(x = x_0 + x_1\), and
  \(t\,J(t^{-1},u;A_0,A_1) = \max(t\norm{u}{0},\norm{u}{1})\).
  For (a), reindex the infimum by \((x_0,x_1) \mapsto (x_1,x_0)\); since \(\alpha\) is
  commutative, this map is a bijection of the set of decompositions of \(x\).
  (b) is the commutativity of \(\max\).
  The special cases are \(t = 2^\nu\), where \((2^\nu)^{-1} = 2^{-\nu}\).
\end{proof}

\begin{lemma}\label{lem:reflect}
  For \(g \colon \Z \to [0,\infty]\) and \(\theta \in \mathbb R\),
  \(\Phi_\theta(\nu \mapsto 2^\nu g_{-\nu}) = \Phi_{1-\theta}(g)\).
\end{lemma}

\begin{proof}
  Since \(2^{-\nu\theta}2^\nu = 2^{\nu(1-\theta)} = 2^{-(-\nu)(1-\theta)}\), the family
  defining the left-hand side is \((h_{-\nu})_{\nu\in\Z}\), where
  \(h_\mu = 2^{-\mu(1-\theta)} g_\mu\) is the family defining \(\Phi_{1-\theta}(g)\). Apply
  Lemma~\ref{lem:lq-facts}(b) with \(\sigma(\nu) = -\nu\).
\end{proof}

In Lean this is \texttt{discretePhiFunctional\_reflect} (\texttt{KMethod.lean}). The weight in
\texttt{discretePhiFunctional} is the real power
\(2^{(-\nu)\cdot\theta}\), while \(2^\nu\) is an integer power. The identity of the weights
follows from \texttt{ENNReal.rpow\_intCast}, \texttt{ENNReal.rpow\_add} (with
\texttt{two\_ne\_zero} and \texttt{ENNReal.ofNat\_ne\_top}) and \texttt{push\_cast; ring} on
the exponents. The reflection is \texttt{eLpNorm\_comp\_measurePreserving} with
\texttt{AEStronglyMeasurable.of\_discrete} and
\texttt{neg\_bijective.measurePreserving\_count .of\_discrete}.

\begin{proposition}\label{prop:swap}
  For \(x \in \alpha\) and \(\theta \in \mathbb R\),
  \begin{enumerate}
    \item[(a)] \(\norm{x}{K}(A_1,A_0;\theta) = \norm{x}{K}(A_0,A_1;1-\theta)\);
    \item[(b)] \(\cJ(x;A_1,A_0;\theta) = \cJ(x;A_0,A_1;1-\theta)\).
  \end{enumerate}
\end{proposition}

\begin{proof}
  (a) By Lemma~\ref{lem:swap-KJ} and Lemma~\ref{lem:reflect} with
  \(g_\mu = K_\mu(x;A_0,A_1)\),
  \[
    \norm{x}{K}(A_1,A_0;\theta) = \Phi_\theta\bigl(\nu \mapsto 2^\nu K_{-\nu}(x;A_0,A_1)\bigr)
    = \Phi_{1-\theta}(g) = \norm{x}{K}(A_0,A_1;1-\theta).
  \]

  (b) For \(u \colon \Z \to \alpha\) let \(w \colon \Z \to \alpha\), \(w_\nu = u_{-\nu}\).
  By Lemma~\ref{lem:swap-KJ}, \(J_\nu(u_\nu;A_1,A_0) = 2^\nu J_{-\nu}(w_{-\nu};A_0,A_1)\).
  So by Lemma~\ref{lem:reflect} with \(g_\mu = J_\mu(w_\mu;A_0,A_1)\), the cost of \(u\) for
  \((A_1,A_0;\theta)\) is \(\Phi_{1-\theta}(g)\), the cost of \(w\) for
  \((A_0,A_1;1-\theta)\). The family \(u\) is admissible if and only if \(w\) is: the support
  of \(w\) is the negative of the support of \(u\), and \(\sum_\nu w_\nu = \sum_\nu u_\nu\)
  because \(\nu \mapsto -\nu\) is a bijection of \(\Z\). Every family \(\Z \to \alpha\) is of
  the form \(w\) for some \(u\), namely for \(u_\nu = w_{-\nu}\). Hence, as \(u\) runs through
  the admissible families, \(w\) runs through all admissible families, and the two infima in
  (b) are infima of the same set of costs.
\end{proof}

\begin{corollary}\label{cor:one-minus-theta}
  For every \(x \in \Delta\), \(\cJ(x) \le 2^{1-\theta}\norm{x}{K}\).
\end{corollary}

\begin{proof}
  The hypotheses are invariant under swapping \(A_0\) and \(A_1\) and replacing \(\theta\) by
  \(1-\theta\): (R) concerns \(\alpha\) alone, both quasinorms are solid,
  \(0 < 1-\theta < 1\), and the definition of \(\Delta\) is symmetric in \(A_0\) and \(A_1\).
  So Proposition~\ref{prop:two-theta} applies to \((A_1,A_0)\) and \(1-\theta\). Together with
  Proposition~\ref{prop:swap}(b) and (a), each taken with \(1-\theta\) in place of \(\theta\),
  it gives
  \[
    \cJ(x;A_0,A_1;\theta) = \cJ(x;A_1,A_0;1-\theta)
    \le 2^{1-\theta}\norm{x}{K}(A_1,A_0;1-\theta) = 2^{1-\theta}\norm{x}{K}(A_0,A_1;\theta).
    \qedhere
  \]
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:main}]
  Since \(2^{\min(\theta,1-\theta)}\) is \(2^\theta\) or \(2^{1-\theta}\)
  (\texttt{min\_choice}), the theorem follows from Proposition~\ref{prop:two-theta} and

\end{proof}

\section{Remarks}

\begin{remark}[Where the hypotheses are used]
  \leavevmode
  \begin{itemize}
    \item (R), \eqref{eq:abs-add} and transitivity of \(\le\): only in
      Proposition~\ref{prop:split-general}.
    \item \(\abs{0} = 0\) and \(0 \le \abs{x}\): only in the Lean form of
      Proposition~\ref{prop:split-general}, through \(\abs{0} \le \abs{w}\) for \(\nu \notin I\).
    \item Solidity of \(A_0, A_1\): only in Lemma~\ref{lem:pointwise}.
    \item \(x \in \Delta\) and \(0 < \theta < 1\): in \(e_{-N}, e_{N-1} \to 0\). Each end
      uses one half of each condition: \(e_{-N}\) uses \(\theta < 1\) and
      \(\norm{x}{1} < \infty\), \(e_{N-1}\) uses \(\theta > 0\) and \(\norm{x}{0} < \infty\).
      Besides, \(\theta > 0\) gives the bound \(2^{-\nu\theta} \le 2^{N\theta}\) in
      Lemma~\ref{lem:termwise}, where any finite bound would do.
    \item The quasi-triangle inequalities of \(A_0\) and \(A_1\): never. For \(\ell^q\), only
      Lemma~\ref{lem:lq-sub} is used. Its constant is \(1\) for every \(q\), which is why the
      constant of Theorem~\ref{thm:main} does not depend on \(q\); the quasi-triangle
      inequality of \(\ell^q\) would contribute a factor \(2^{1/q-1}\) for \(q < 1\).
  \end{itemize}
  For \(q = 0\), Mathlib's \texttt{eLpNorm} is identically \(0\). The family with \(u_0 = x\)
  and \(u_\nu = 0\) for \(\nu \ne 0\) (\texttt{Pi.single 0 x}) is admissible, so
  \(\cJ(x) = 0\) and the inequality is trivial. This case is excluded from the proof above
  because there \(r = 0\).
\end{remark}

\begin{remark}[Why \(x \in \Delta\) is necessary]
  Let \(q \neq 0\). If \(\cJ(x) < \infty\), some admissible \(\ell^q\)-norm is finite, so each
  \(J_\nu(u_\nu)\) is finite and each \(u_\nu\) lies in \(\Delta\). Since \(\Delta\) is an
  additive submonoid (by the quasi-triangle inequality with finite constants), the finite sum
  \(x\) lies in \(\Delta\). Hence \(\cJ = \infty\) outside \(\Delta\), whereas
  \(\norm{x}{K}\) may be finite there.
\end{remark}
